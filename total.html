return null; } function _ymd(d){ return d.getFullYear()+'-'+('0'+(d.getMonth()+1)).slice(-2)+'-'+('0'+d.getDate()).slice(-2); } function _weekMonday(d){ // 그 주 월요일 var day = (d.getDay()+6)%7; // 월=0 var m = new Date(d.getFullYear(), d.getMonth(), d.getDate()-day); }); } catch(e){ Logger.log('월종결 로드 실패: '+e); } return out; } function buildTotalData() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(TOTAL_SRC_SHEET); if (!sh) { throw new Error('탭 없음: ' + TOTAL_SRC_SHEET); } var values = sh.getDataRange().getValues(); if (values.length < 2) { throw new Error('총매출_전사 데이터 없음'); } // 헤더에서 컬럼 인덱스 찾기 (순서 바뀌어도 안전) var H = values[0].map(function(x){ return String(x || '').trim(); }); function col(name){ for (var i=0;i= 0; }; // 집계 컨테이너 var tot = { sales:0, ord:0, set:0 }; var web = { sales:0, ord:0, set:0, cxa:0, cxc:0 }; var sang = { sales:0, ord:0, set:0, cxa:0, cxc:0 }; var brMap = {}; // 지점 → {sales,ord,set,cxa,cxc} var perMap = {}; // 상담개인(B) → {team(H지점), sales,ord,set,cxa} var divpMap = {}; // 사업개인(F) → {sales,ord,set,cxa} var cancelAmt = 0, cancelCnt = 0; function ensure(map, key, extra){ if (!map[key]) { map[key] = { sales:0, ord:0, set:0, cxa:0, cxc:0 }; if (extra) for (var k in extra) map[key][k]=extra[k]; } return map[key]; } for (var r = 1; r < values.length; r++) { var row = values[r]; var b = String(row[iB] || '').trim(); if (!b) continue; var salesRaw = row[iSales]; var sales = _num(salesRaw); // 매출(천원) var note = iNote>=0 ? String(row[iNote]||'') : ''; var hRaw = row[iH]; var fRaw = iF>=0 ? String(row[iF]||'').trim() : ''; var isOrder = sales > 0; var isSet = true; // 시트에 올라온 행 = 세팅 이상 (결보 포함). 세팅수는 전체 행 기준. // 취소 추출 (특이사항 "숫자취소" 패턴) var cxAmt = 0; var m = note.match(/(\d+)\s*취소/); if (m) cxAmt = parseInt(m[1], 10); // 전사 tot.set++; if (isOrder) { tot.sales += sales; tot.ord++; } if (isWeb(b)) { web.set++; if (isOrder) { web.sales += sales; web.ord++; } if (cxAmt) { web.cxa += cxAmt; web.cxc++; } } else { sang.set++; if (isOrder) { sang.sales += sales; sang.ord++; } if (cxAmt) { sang.cxa += cxAmt; sang.cxc++; } // 지점(branches) — 웹마 제외, 인투본사(웹마 결보 자리) 제외 var br = _h2branch(hRaw); if (br && br !== '인투본사') { var bo = ensure(brMap, br); bo.set++; if (isOrder) { bo.sales += sales; bo.ord++; } if (cxAmt) { bo.cxa += cxAmt; bo.cxc++; } } // 상담개인(persons) — B별 var po = ensure(perMap, b, { team: br || '' }); po.set++; if (isOrder) { po.sales += sales; po.ord++; } if (cxAmt) po.cxa += cxAmt; // 사업개인(divpersons) — F별 if (fRaw) { var dpo = ensure(divpMap, fRaw); dpo.set++; if (isOrder) { dpo.sales += sales; dpo.ord++; } if (cxAmt) dpo.cxa += cxAmt; } } if (cxAmt) { cancelAmt += cxAmt; cancelCnt++; } } // 천원 → 원 변환 + 정렬/구조화 function won(n){ return Math.round(n * 1000); } function rate(o,s){ return s ? }; // 기간 필터로 상담(B)/사업(F) 개인 매출·세팅 집계 → 소속 포함 function aggPersons(dateFilter, useF) { var m = {}; // key → {sales,set,gb,team} allRows.forEach(function(r){ if (!r.date || !dateFilter(r.date)) return; if (isWebRow(r.b)) return; // 웹마 제외 var key = useF ? r.f : r.b; if (!key) return; if (!m[key]) m[key] = { name:key, sales:0, set:0, gb:0, team: _h2branch(r.h) }; m[key].set++; if (r.sales > 0) m[key].sales += r.sales; else if (r.gb) m[key].gb++; // 결보 카운트 (aggPersons 호출 전 r.gb 세팅) if (!m[key].team && r.h) m[key].team = _h2branch(r.h); }); return m; } function topBy(m, field, n) { var arr = []; for (var k in m){ var v=m[k]; arr.push({ name:v.name, team:v.team||'', sales:won(v.sales), set:v.set, gb:v.gb||0 });
